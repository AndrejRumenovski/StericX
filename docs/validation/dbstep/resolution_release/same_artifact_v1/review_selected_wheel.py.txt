"""Bind four actual Windows installations to one unchanged candidate wheel."""

import gzip
import hashlib
import json
import re
import shutil
import subprocess
from pathlib import Path

RAW = Path(__file__).resolve().parent
ROOT = RAW.parents[3]
OUTPUT = ROOT / "docs/validation/dbstep/resolution_release/same_artifact_v1"
SOURCE = "03557ed5279abf8487012ddb39fab019f0538416"
WORKFLOW = "6c2eddfadc3329b6ddf2e40dc2eb981ef865e4f2"
DIGEST = "900fa8616446459cfd544edb5633c167b20bae0d0d0e01d19c78c9b0a6e69864"


def record(path):
    return {
        "path": str(path.resolve()),
        "sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
        "bytes": path.stat().st_size,
    }


def main():
    OUTPUT.mkdir(parents=True, exist_ok=False)
    run = json.loads((RAW / "run_final.json").read_text())
    assert run["databaseId"] == 41507681984
    assert run["headSha"] == WORKFLOW
    assert run["status"] == "completed" and run["conclusion"] == "success"
    assert len(run["jobs"]) == 4
    assert all(j["conclusion"] == "success" for j in run["jobs"])
    workflow = subprocess.check_output(
        ["git", "show", f"{WORKFLOW}:.github/workflows/python-rc-artifact.yml"],
        cwd=ROOT,
    )
    assert workflow == (RAW / "workflow.yml.txt").read_bytes()
    assert workflow == (ROOT / ".github/workflows/python-rc-artifact.yml").read_bytes()
    wheel = (
        RAW.parent
        / "ci_raw/artifacts/wheel-windows-latest-py3.11/stericx-0.4.0rc1-cp311-abi3-win_amd64.whl"
    )
    artifact = record(wheel)
    assert artifact["sha256"] == DIGEST
    receipts = []
    for minor in ["3.11", "3.12", "3.13", "3.14"]:
        directory = RAW / "artifacts" / f"frozen-windows-wheel-py{minor}"
        installed_path = directory / "installed.json"
        installed = json.loads(installed_path.read_text())
        assert installed["source_commit"] == SOURCE
        assert installed["workflow_commit"] == WORKFLOW
        assert installed["source_run"] == 41505795041
        assert installed["wheel_sha256"] == DIGEST
        assert installed["wheel_bytes"] == artifact["bytes"]
        assert installed["python_minor"] == minor and installed["python"].startswith(
            minor + "."
        )
        assert installed["platform"].startswith("Windows")
        assert installed["pip_check_returncode"] == 0
        assert (
            installed["installed"]["version"]
            == installed["installed"]["metadata_version"]
            == "0.4.0rc1"
        )
        assert "site-packages" in installed["installed"]["module"]
        test_path = directory / "tests.log"
        text = test_path.read_text()
        assert re.search(r"\b71 passed in ", text)
        assert not re.search(r"\b\d+ (failed|error|errors|skipped)\b", text)
        check_path = directory / "pip_check.log"
        assert "No broken requirements found." in check_path.read_text()
        receipts.append(
            {
                "python_minor": minor,
                "installed": installed,
                "installed_record": record(installed_path),
                "tests_record": record(test_path),
                "pip_check_record": record(check_path),
                "tests_passed": 71,
            }
        )
    records = [record(p) for p in sorted(RAW.rglob("*")) if p.is_file()]
    for source in sorted(RAW.rglob("*")):
        if not source.is_file():
            continue
        relative = source.relative_to(RAW)
        destination = OUTPUT / relative
        destination.parent.mkdir(parents=True, exist_ok=True)
        if source.name == "full.log":
            with destination.with_suffix(".log.gz").open("wb") as stream:
                with gzip.GzipFile(
                    fileobj=stream, mode="wb", filename="", mtime=0
                ) as compressed:
                    compressed.write(source.read_bytes())
        else:
            shutil.copyfile(source, destination)
    result = {
        "passed": True,
        "source_commit": SOURCE,
        "workflow_commit": WORKFLOW,
        "run": run,
        "artifact_sha256": DIGEST,
        "artifact": artifact,
        "python_versions": ["3.11", "3.12", "3.13", "3.14"],
        "tests_each": 71,
        "receipts": receipts,
        "records": records,
        "failures": [],
        "publication": False,
    }
    (OUTPUT / "same_artifact_review.json").write_text(
        json.dumps(result, indent=2) + "\n"
    )
    manifest = {
        str(p.relative_to(OUTPUT)): record(p)["sha256"]
        for p in sorted(OUTPUT.rglob("*"))
        if p.is_file()
    }
    (OUTPUT / "manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")
    print(
        json.dumps(
            {
                "passed": True,
                "tests_each": 71,
                "python_versions": result["python_versions"],
                "artifact_sha256": DIGEST,
                "raw_records": len(records),
            },
            indent=2,
        )
    )


if __name__ == "__main__":
    main()

import hashlib
import json
import os
import subprocess
import time
from datetime import UTC, datetime
from pathlib import Path

ROOT = Path.cwd()
OUT = Path(__file__).resolve().parent
LOGS = ROOT / "docs/validation/dbstep/resolution_validation"


def record(path):
    return {
        "path": str(path),
        "bytes": path.stat().st_size,
        "sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
    }


def snapshot():
    names = subprocess.check_output(
        ["git", "ls-files", "--cached", "--others", "--exclude-standard"], text=True
    ).splitlines()
    return [
        record(ROOT / name)
        for name in sorted(set(names))
        if not name.startswith("docs/media/")
        and (
            Path(name).suffix in {".py", ".pyi", ".rs"}
            or Path(name).name
            in {"Cargo.toml", "Cargo.lock", "pyproject.toml", "uv.lock", ".ruff.toml"}
        )
    ]


before = snapshot()
result = {
    "kind": "final_source_stable_style_research_checks",
    "started_utc": datetime.now(UTC).isoformat(),
    "controller": record(Path(__file__).resolve()),
    "sources_before": before,
    "runs": [],
    "complete": False,
}
env = dict(os.environ, OPENBLAS_NUM_THREADS="1", OMP_NUM_THREADS="1", MKL_NUM_THREADS="1", RAYON_NUM_THREADS="2")
commands = [
    ("final_ruff", [str(ROOT / ".venv/bin/ruff"), "check", ".", "--extend-exclude", "docs/media"]),
    ("final_python_format", [str(ROOT / ".venv/bin/ruff"), "format", "--check", ".", "--extend-exclude", "docs/media"]),
    ("final_python_tests", [str(ROOT / ".venv/bin/python"), "-m", "unittest", "discover", "-s", "tests", "-p", "test_*.py"]),
]
for name, argv in commands:
    stdout, stderr = LOGS / f"{name}.stdout.log", LOGS / f"{name}.stderr.log"
    start = time.monotonic()
    with stdout.open("xb") as out, stderr.open("xb") as err:
        completed = subprocess.run(argv, cwd=ROOT, env=env, stdout=out, stderr=err)
    result["runs"].append({"name": name, "argv": argv, "cwd": str(ROOT), "seconds": time.monotonic()-start, "returncode": completed.returncode, "stdout": record(stdout), "stderr": record(stderr)})
    (OUT / "journal.json").write_text(json.dumps(result, indent=2)+"\n")
    print(name, completed.returncode, flush=True)
    if completed.returncode:
        raise SystemExit(completed.returncode)
after = snapshot()
result.update(sources_after=after, sources_unchanged=before == after, complete=before == after, completed_utc=datetime.now(UTC).isoformat())
(OUT / "journal.json").write_text(json.dumps(result, indent=2)+"\n")
if before != after:
    raise SystemExit("Scientific/research sources changed during checks")

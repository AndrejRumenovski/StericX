def storage_counts(index):
    local = [x for x in index["blobs"].values() if x.get("kind") != "publication_chunk"]
    return {"files": len(index["inventory"]), "unique_chunks": len(index["blobs"]),
            "shared_files": sum("shared" in row for row in index["inventory"]),
            "publication_files": sum(row.get("shared", {}).get("kind") == "publication_file"
                                     for row in index["inventory"]),

def reconstructed(package, index, row, shared_root):
    providers = providers_for(shared_root, index["providers"])
    if "shared" in row:
        locator = row["shared"]
        if locator["kind"] == "publication_file":
            yield from providers.file(locator, row["sha256"], row["bytes"])
        else:
            yield decode_shared(shared_root, locator)
    else:
        for key in row["chunks"]:
            yield shared_chunk(package, key, index["blobs"][key], providers)


def load_bound(package):
    first = record(package / "manifest.json")
    manifest, index = load_package(package)
    pinned = {name: record(within(package, name)) for name in
              ["manifest.json", *[manifest[field]["path"] for field in ("index", "controller", "plan")],

    entries = {row["path"]: row for row in index["inventory"]}
    for item in (plan["spec"], plan["controller"], *plan["helper_sources"]):
        relative = Path(item["path"]).relative_to(plan["repo"]).as_posix()
        if relative not in entries or any(entries[relative][key] != item[key] for key in ("bytes", "sha256")):
            raise ValueError("Original specification/controller/helper absent")
    if {k for row in index["inventory"] for k in row.get("chunks", [])} != set(index["blobs"]):

        raise ValueError("Published storage counts differ")

        raise ValueError("Published helper bindings differ from plan")
    return manifest, index, pinned


def check_pins(package, pinned, providers):
    for name, expected in pinned.items():
        if record(within(package, name)) != expected:
            raise ValueError("Publication metadata changed during operation")
    providers.check_pins()


def verify_context(args):
    package = args.package.resolve()
    manifest, index, pinned = load_bound(package)
    providers = provider_pins(args.shared_root, index["providers"])
    for key, blob in index["blobs"].items():
        shared_chunk(package, key, blob, providers)
    entries = {row["path"]: row for row in index["inventory"]}
    seen = set()
    for row in entries.values():
        identity = (row["sha256"], row["bytes"], json.dumps(row.get("shared"), sort_keys=True),
                    tuple(row.get("chunks", [])))
        if identity in seen:
            continue
        digest = hashlib.sha256()
        size = 0
        for raw in reconstructed(package, index, row, args.shared_root):
            digest.update(raw)
            size += len(raw)
        if size != row["bytes"] or digest.hexdigest() != row["sha256"]:
            raise ValueError(f"Restored file differs: {row['path']}")
        seen.add(identity)
    actual = coverage(index, lambda relative: b"".join(reconstructed(package, index, entries[relative], args.shared_root)))
    if actual != index["coverage"] or actual != manifest["coverage"]:
        raise ValueError("Recomputed coverage differs")
    check_pins(package, pinned, providers)
    result = {"verified": True, "files": len(entries), "unique_file_reconstructions": len(seen),
              "counts": storage_counts(index), "coverage": actual, "manifest": pinned["manifest.json"],
              "complete_original_plan_inventory_verified": True,
              "scope": "Lossless byte and declared coverage verification; no scientific or performance rerun."}
    return result, index, pinned, providers


def verify(args):
    return verify_context(args)[0]


def restore(args):
    result, index, pinned, providers = verify_context(args)
    package = args.package.resolve()
    args.output.mkdir(parents=True, exist_ok=False)
    for row in index["inventory"]:
        path = within(args.output.resolve(), row["path"])
        path.parent.mkdir(parents=True, exist_ok=True)
        with path.open("xb") as stream:
            for raw in reconstructed(package, index, row, args.shared_root):
                stream.write(raw)

            raise ValueError("Restored file differs after write")
    check_pins(package, pinned, providers)
    result.update(restored_root=str(args.output.resolve()), restored_from_verified_in_memory_index=True,
                  external_links_not_created=index["external_links"], omissions=index["omissions"])
    write_new(args.output / "RESTORATION_RECEIPT.json", json_bytes(result))
    return result

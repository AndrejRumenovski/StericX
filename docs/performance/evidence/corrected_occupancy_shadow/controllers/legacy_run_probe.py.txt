def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, indent=2, sort_keys=True)
        stream.write('\n')


def identity(path):
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': sha(path)}


def replace_once(text, old, new):
    assert text.count(old) == 1, old[:120]
    return text.replace(old, new)

struct ProbeRow { start: usize, end: usize, finite: bool, minimum: f32, maximum: f32 }

fn probe_rows(points: &[Vec3]) -> Vec<ProbeRow> {
    let mut rows = Vec::new(); let mut start = 0;
    while start < points.len() {
        let mut end = start + 1;
        while end < points.len() && points[end].x.to_bits() == points[start].x.to_bits()
            && points[end].y.to_bits() == points[start].y.to_bits() { end += 1; }
        let finite = points[start..end].iter().all(|p| p.is_finite());
        let minimum = points[start..end].iter().map(|p| p.z).fold(f32::INFINITY, f32::min);
        let maximum = points[start..end].iter().map(|p| p.z).fold(f32::NEG_INFINITY, f32::max);
        rows.push(ProbeRow { start, end, finite, minimum, maximum }); start = end;
    }
    rows
}

// 0: conservative fallback, 1: atom within the actual row z range,
// 2: closest actual endpoint hits, 3: closest actual endpoint misses.
fn probe_endpoint(row: &ProbeRow, atom: &AlignedAtom, xy: f32) -> u8 {
    if !row.finite || !atom.position.is_finite() || !atom.radius_squared.is_finite()
        || !xy.is_finite() { return 0; }
    let endpoint = if atom.position.z < row.minimum { row.minimum }
        else if atom.position.z > row.maximum { row.maximum } else { return 1; };
    let dz = endpoint - atom.position.z;
    if xy + dz * dz <= atom.radius_squared { 2 } else { 3 }
}

fn probe_bits(v: &OccupiedVolumes) -> Vec<u32> {
    let mut bits = vec![v.buried_volume.to_bits(), v.near_vbur.to_bits(), v.far_vbur.to_bits()];
    bits.extend(v.quadrants.map(f32::to_bits)); bits.extend(v.octants.map(f32::to_bits)); bits
}

type ProbeWriter = std::sync::Mutex<Option<std::io::BufWriter<std::fs::File>>>;
static PROBE_WRITER: std::sync::OnceLock<ProbeWriter> = std::sync::OnceLock::new();

fn probe_record(value: serde_json::Value) {
    use std::io::Write;
    let output = PROBE_WRITER.get_or_init(|| std::sync::Mutex::new(
        std::env::var_os("STERICX_ROW_RANGE_PROBE_PATH").map(|path| {
            std::io::BufWriter::new(std::fs::OpenOptions::new().write(true).create_new(true)
                .open(path).expect("fresh diagnostic sidecar"))
        })));
    if let Some(writer) = output.lock().unwrap().as_mut() {
        serde_json::to_writer(&mut *writer, &value).unwrap(); writer.write_all(b"\n").unwrap();
    }
}

pub fn row_range_probe_finish() {
    use std::io::Write;
    if let Some(writer) = PROBE_WRITER.get()
        && let Some(writer) = writer.lock().unwrap().as_mut() { writer.flush().unwrap(); }
}

pub fn row_range_probe_conformer(molecule: &Molecule, donor: usize, reference: usize,
    config: BuriedVolumeConfig, source: &str, conformer: usize) {
    let (neighbors, center) = donor_geometry(molecule, donor, reference, config).unwrap();
    let points = integration_grid(config);
    for (orientation, plane) in neighbors.into_iter().enumerate() {
        let basis = coordinate_basis(molecule, donor, plane, center).unwrap();
        let atoms = aligned_atoms(molecule, center, basis, config).unwrap();
        let context = serde_json::json!({"source":source,"conformer":conformer,
            "orientation":orientation,"plane":plane,"atoms":atoms.len(),"points":points.len()});
        let _ = probe_instrumented_occupied_volumes(&points, &atoms, config.sphere_radius, &context);
    }
}

def instrument(original):

    function=original[start:end]
    (ROOT/'current_occupied_volumes.rs').write_text(function)
    function=replace_once(function,'fn occupied_volumes(sphere: &[Vec3], atoms: &[AlignedAtom], sphere_radius: f32)',
        'fn probe_instrumented_occupied_volumes(sphere: &[Vec3], atoms: &[AlignedAtom], sphere_radius: f32, context: &serde_json::Value)')
    function=replace_once(function,'    struct RowCandidate {','''    let rows = probe_rows(sphere);
    let mut counts = ProbeCounts::default();
    let mut row_cursor = 0;
    let mut row_lengths = std::collections::BTreeMap::<usize,u64>::new();

    let mut witnesses = Vec::new();
    // Eager model is computed separately; these atoms are NOT added to the
    // actual lazy cache. It is an upper-work model, not an accepted algorithm.
    for row in &rows {
        *row_lengths.entry(row.end-row.start).or_default() += 1;
        for atom in atoms {
            counts.eager_xy_preparations += 1;
            let dx=sphere[row.start].x-atom.position.x;
            let dy=sphere[row.start].y-atom.position.y;
            let xy=dx*dx+dy*dy;
            if xy > atom.radius_squared { counts.eager_xy_rejections += 1; continue; }
            match probe_endpoint(row,atom,xy) {
                0 => counts.eager_nonfinite_fallback += 1,
                1 => counts.eager_inside_z_range += 1,
                2 => counts.eager_endpoint_checks += 1,
                3 => {counts.eager_endpoint_checks += 1;counts.eager_endpoint_rejections += 1;},
                _ => unreachable!(),
            }
        }
    }
    struct RowCandidate {
        atom_index: usize,
        shadow_skip: bool,''')
    function=replace_once(function,'    for point in sphere {','''    for (point_index, point) in sphere.iter().enumerate() {
        counts.point_visits += 1;
        let mut hit_atom = atoms.len();''')
    function=replace_once(function,'            row_xy = Some(xy);','''            if row_xy.is_some() { row_cursor += 1; }
            counts.rows += 1;

                eliminated_by_atom[candidate.atom_index] += 1;

                    "radius_squared_bits":candidate.radius_squared.to_bits()})); }

            accepted''')
    function=replace_once(function,'            let atom = &atoms[next_atom];','''            counts.xy_preparations += 1;
            let atom = &atoms[next_atom];''')
    function=replace_once(function,'''            if xy_squared > atom.radius_squared {
                continue;
            }
            candidates.push(RowCandidate {''','''            if xy_squared > atom.radius_squared {
                counts.xy_rejections += 1;

            }
            let endpoint = probe_endpoint(&rows[row_cursor],atom,xy_squared);
            match endpoint {
                0 => counts.nonfinite_fallback_lazy += 1,
                1 => counts.inside_z_range_lazy += 1,
                2 => counts.endpoint_checks_lazy += 1,
                3 => {counts.endpoint_checks_lazy += 1;counts.endpoint_rejections_lazy += 1;},

        "original_first_hits_by_atom_or_miss":first_hits_by_atom,

        "matches_unmodified_current_function":true}));

    (ROOT/'instrumented_occupied_volumes.rs').write_text(function)
    (ROOT/'instrumented_loop.patch').write_text(''.join(difflib.unified_diff(

def compile_probe():
    target=ROOT/'target'
    commands=[['cargo','build','--offline','--locked','--release','--bin','stericx',
        '--manifest-path',str(SOURCE/'Cargo.toml'),'--target-dir',str(target)],
        ['cargo','test','--offline','--locked','--release','--lib',
        '--manifest-path',str(SOURCE/'Cargo.toml'),'--target-dir',str(target),
        'row_range_shadow_tests::row_range_shadow_preserves_boundaries_nonfinite_and_arbitrary_order']]
    records=[]
    for i,command in enumerate(commands):
        with (ROOT/f'build_{i}.stdout').open('xb') as out,(ROOT/f'build_{i}.stderr').open('xb') as err:
            result=subprocess.run(command,cwd=SOURCE,stdout=out,stderr=err)
        records.append({'argv':command,'returncode':result.returncode,
            'stdout':identity(ROOT/f'build_{i}.stdout'),'stderr':identity(ROOT/f'build_{i}.stderr')})
        if result.returncode:
            save(ROOT/'build_failed.json',records)
            raise RuntimeError('Probe build/test failed')
    save(ROOT/'build_manifest.json',{'commands':records,'binary':identity(target/'release/stericx'),
        'rustc':subprocess.check_output(['rustc','-vV'],text=True),'cargo':subprocess.check_output(['cargo','-V'],text=True)})


def summarize(path):
    totals=Counter();row_lengths=Counter();firsthits=Counter();per_geometry={};orientations=[]
    records=0
    for line in path.open():
        row=json.loads(line);records+=1
        assert row['matches_unmodified_current_function']
        totals.update(row['counts']);row_lengths.update(row['row_length_histogram'])
        for i,count in enumerate(row['original_first_hits_by_atom_or_miss']):
            firsthits['miss' if i==row['context']['atoms'] else str(i)]+=count
        key=(row['context']['source'],row['context']['conformer'])
        per_geometry[key]=True
        c=row['counts'];eliminated=c['eliminated_cached_z_tests']+c['eliminated_new_z_tests']
        orientations.append({'context':row['context'],'eliminated_tests':eliminated,
            'added_endpoint_checks':c['endpoint_checks_lazy'],
            'net_z_predicate_count_reduction':eliminated-c['endpoint_checks_lazy']})
    ztests=totals['cached_z_tests']+totals['new_candidate_z_tests']
    eliminated=totals['eliminated_cached_z_tests']+totals['eliminated_new_z_tests']
    added=totals['endpoint_checks_lazy']
    assert totals['asserted_eliminated_tests_are_misses']==eliminated
    orientations.sort(key=lambda v:v['net_z_predicate_count_reduction'])
    return {'orientations':records,'geometries':len(per_geometry),'totals':dict(totals),
        'current_z_predicates':ztests,'shadow_eliminable_z_predicates':eliminated,
        'shadow_lazy_added_endpoint_predicates':added,'shadow_lazy_net_z_predicate_change':added-eliminated,
        'shadow_lazy_net_z_predicate_reduction_fraction':(eliminated-added)/ztests,
        'shadow_lazy_z_predicates_remaining_plus_endpoints':ztests-eliminated+added,
        'eager_upper_model_xy_preparations':totals['eager_xy_preparations'],
        'eager_upper_model_added_endpoint_checks':totals['eager_endpoint_checks'],
        'eager_upper_model_extra_xy_preparations':totals['eager_xy_preparations']-totals['xy_preparations'],
        'row_length_histogram':dict(row_lengths),'original_first_hit_histogram':dict(firsthits),
        'orientations_net_z_tests_increased':sum(v['net_z_predicate_count_reduction']<0 for v in orientations),
        'lowest_net_reduction_examples':orientations[:5],'highest_net_reduction_examples':orientations[-5:],
        'limitations':['Operation counts are not machine instructions or performance predictions.',
            'The probe always executes original current predicates and adds shadow work; it is excluded from native timing claims.',
            'Row min/max scans, finite checks, branches, vector accesses and instrumentation are not represented by the z-predicate net count.',
            'The eager upper-work model visits every row/atom; it is not the lazy candidate algorithm.',
            'Repeated throughput fixtures do not represent 10000 distinct chemistries.']}


def run():
    workload_path=PROFILE/'workloads/manifest.json'
    manifest=json.loads(workload_path.read_text());verify_manifest(manifest)
    fresh=PROFILE/'runs/current_audited_b515c4f_20260921_native_t1/summary.json'
    native_reference=json.loads(fresh.read_text())
    summary={'purpose':__doc__,'scientific_findings':'Preserved current audited limitations, not a corrected scientific baseline.',
        'workload_manifest':identity(workload_path),'fresh_native_summary':identity(fresh),
        'source_manifest':identity(ROOT/'source_manifest.json'),'build_manifest':identity(ROOT/'build_manifest.json'),
        'results':{},'no_performance_claim':True}
    env={**os.environ,**CONTROLLED_ENV,'RAYON_NUM_THREADS':'1'};env.pop('STERICX_PROFILE_PATH',None)
    for name in ['conformers_56','descriptors_10000']:
        workload=manifest['workloads'][name]
        for label,binary in [('current_native',CAPTURE/'bin/native/stericx'),('probe',ROOT/'target/release/stericx')]:
            directory=ROOT/name/label;directory.mkdir(parents=True,exist_ok=False)
            argv=[str(binary),*[v.replace('{run_dir}',str(directory)) for v in workload['argv']]]
            runenv=dict(env)
            if label=='probe':runenv['STERICX_ROW_RANGE_PROBE_PATH']=str(directory/'counts.jsonl')
            else:runenv.pop('STERICX_ROW_RANGE_PROBE_PATH',None)
            save(directory/'command.json',{'argv':argv,'cwd':str(REPO),'environment':{k:runenv.get(k) for k in [*CONTROLLED_ENV,'RAYON_NUM_THREADS','STERICX_ROW_RANGE_PROBE_PATH']},'binary':identity(binary)})
            print(name,label,'running (diagnostic only)',flush=True)
            with (directory/'stdout.txt').open('xb') as out,(directory/'stderr.txt').open('xb') as err:
                result=subprocess.run(argv,cwd=REPO,env=runenv,stdout=out,stderr=err)
            assert result.returncode==0,(name,label,result.returncode)
            fingerprint=check_result(workload,directory)
            assert fingerprint==native_reference['results'][name]['fingerprints']
